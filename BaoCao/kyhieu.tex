\addchap{DANH MỤC KÝ HIỆU VÀ CHỮ VIẾT TẮT}

\section*{Chữ viết tắt}
\begin{longtable}{@{}p{2.6cm}p{12.2cm}@{}}
\toprule
\textbf{Viết tắt} & \textbf{Ý nghĩa} \\ \midrule
\endhead
PIR  & Passive InfraRed -- hồng ngoại thụ động \\
IR   & InfraRed -- bức xạ hồng ngoại \\
JFET & Junction Field-Effect Transistor -- tranzito hiệu ứng trường mối nối \\
op-amp & Operational amplifier -- khuếch đại thuật toán \\
RRIO & Rail-to-Rail Input/Output -- vào/ra sát hai đường nguồn \\
ADC  & Analog-to-Digital Converter -- bộ biến đổi tương tự -- số \\
MCU  & Microcontroller Unit -- vi điều khiển \\
HPF / LPF & High-Pass / Low-Pass Filter -- bộ lọc thông cao / thông thấp \\
EMI  & Electromagnetic Interference -- nhiễu điện từ \\
PZT  & Lead Zirconate Titanate -- gốm chì zirconat titanat \\
SK   & Sallen-Key (bộ lọc) \quad hoặc \quad Kịch bản (mô phỏng) \\
NEP  & Noise Equivalent Power -- công suất tương đương nhiễu \\
\bottomrule
\end{longtable}

\section*{Ký hiệu}
\begin{longtable}{@{}p{2.6cm}p{9.6cm}p{2.4cm}@{}}
\toprule
\textbf{Ký hiệu} & \textbf{Ý nghĩa} & \textbf{Đơn vị} \\ \midrule
\endhead
$\Phi$, $P$       & Công suất bức xạ hồng ngoại tới phần tử & W, µW \\
$\eta$            & Hệ số hấp thụ hiệu dụng (gồm truyền qua cửa sổ lọc) & -- \\
$T$, $\Delta T$   & Nhiệt độ, độ tăng nhiệt độ của phần tử & K \\
$H$ ($C_{th}$)    & Nhiệt dung của phần tử & J/K \\
$G$ ($G_{th}$)    & Độ dẫn nhiệt từ phần tử ra môi trường & W/K \\
$\tau_T$, $\tau_E$ & Hằng số thời gian nhiệt, hằng số thời gian điện & s \\
$p$               & Hệ số nhiệt điện của vật liệu & C/(m$^2$·K) \\
$A$               & Diện tích một phần tử & m$^2$ \\
$d$               & Bề dày phần tử & m \\
$\varepsilon_r$   & Hằng số điện môi tương đối & -- \\
$C_E$             & Điện dung một phần tử & F \\
$R_G$             & Điện trở cổng (gate) của FET & $\Omega$ \\
$i_p$             & Dòng nhiệt điện & A \\
$R_V$             & Độ nhạy điện áp (responsivity) & V/W \\
$V_S$             & Điện áp chân S (source) của cảm biến & V \\
$\VREF$           & Điện áp tham chiếu (đất ảo) & V \\
$K$, $K_1$, $K_2$ & Hệ số khuếch đại toàn mạch, tầng 1, tầng 2 & V/V, dB \\
$f_0$, $Q$        & Tần số riêng và hệ số phẩm chất của bộ lọc bậc 2 & Hz, -- \\
$f_c$             & Tần số cắt $-3$~dB & Hz \\
\bottomrule
\end{longtable}
